\section{任务与数据}

\begin{frame}{任务定义：KiTS23 四类分割}
  \begin{columns}[T]
    \column{0.56\textwidth}
      \begin{block}{输入与输出}
        给定腹部 CT axial slice $x\in\mathbb{R}^{C\times256\times256}$，预测逐像素标签：
        \[
          y_i\in\{0,1,2,3\}
          =\{\text{背景},\text{肾脏},\text{肿瘤},\text{囊肿}\}.
        \]
        评价时分别计算 kidney、tumor、cyst 的 Dice / IoU，并报告三类前景平均值。
      \end{block}
      \begin{alertblock}{任务难点}
        肾脏区域较大且形态稳定；肿瘤和囊肿面积小、类别不均衡强，且 2D 切片缺少跨层连续性，因此小病灶最容易漏检。
      \end{alertblock}

    \column{0.40\textwidth}
      \begin{block}{数据划分}
        \centering
        \setlength{\tabcolsep}{3pt}
        \begin{tabular}{lrrr}
          \toprule
          划分 & 病例 & 切片 & 前景切片 \\
          \midrule
          训练 & 70 & 6950 & 5585 \\
          验证 & 15 & 1624 & 1323 \\
          测试 & 15 & 1151 & 901 \\
          \bottomrule
        \end{tabular}
      \end{block}
      \begin{exampleblock}{预处理}
        HU window $c=40,w=400$；归一化到 $[0,1]$；resize 到 $256\times256$；按病例划分防止数据泄漏。
      \end{exampleblock}
  \end{columns}
\end{frame}

\section{方法设计}

\begin{frame}{整体实验设计}
  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{exampleblock}{判别式分割}
        直接学习从 CT 到 mask 的映射：
        \[
          \hat y=f_\theta(x).
        \]
        本项目使用 U-Net、SegResNet 和 2.5D U-Net 作为基线。
      \end{exampleblock}
    \column{0.48\textwidth}
      \begin{alertblock}{生成式分割}
        将 mask 看作随机变量，学习条件分布：
        \[
          p_\theta(y\mid x).
        \]
        本项目实现 conditional VAE 与 conditional diffusion。
      \end{alertblock}
  \end{columns}
  \vspace{0.5em}
  \begin{block}{统一评价}
    所有模型在同一 axial test split 上评估，报告 Mean Dice、Mean IoU、各类别 Dice 和单张图像推理时间。
  \end{block}
\end{frame}

\begin{frame}{评价指标：Dice、IoU 与推理时间}
  \begin{columns}[T]
    \column{0.50\textwidth}
      \begin{block}{前景 Dice}
        对类别 $c$：
        \[
          \mathrm{Dice}_c
          =
          \frac{2|\hat Y_c\cap Y_c|}
          {|\hat Y_c|+|Y_c|}.
        \]
        本项目主指标为三类前景平均：
        \[
          \mathrm{MeanDice}
          =
          \frac{D_{\mathrm{kidney}}+D_{\mathrm{tumor}}+D_{\mathrm{cyst}}}{3}.
        \]
      \end{block}
    \column{0.46\textwidth}
      \begin{exampleblock}{其他报告指标}
        \begin{itemize}
          \item \model{Mean IoU}：前景三类 IoU 平均；
          \item \model{Kidney/Tumor/Cyst Dice}：分析各器官表现；
          \item \model{ms/img}：单张 axial slice 推理时间；
          \item \model{sample steps}：diffusion 去噪采样步数。
        \end{itemize}
      \end{exampleblock}
      \begin{alertblock}{注意}
        Mean Dice 和 Mean IoU 不包含 background，避免大面积背景使分数虚高。
      \end{alertblock}
  \end{columns}
\end{frame}

\section{判别式模型}

\begin{frame}{U-Net 与 SegResNet：判别式基线}
  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{U-Net}
        \begin{itemize}
          \item 外部结构：encoder--decoder；
          \item 长 skip connection：连接 encoder 与 decoder；
          \item 局部模块：普通两层卷积块。
        \end{itemize}
      \end{block}
    \column{0.48\textwidth}
      \begin{exampleblock}{SegResNet}
        \begin{itemize}
          \item 外部结构：同样是 encoder--decoder；
          \item 长 skip connection：同样保留多尺度细节；
          \item 局部模块：将普通卷积块替换为 residual block。
        \end{itemize}
      \end{exampleblock}
  \end{columns}
\end{frame}

\begin{frame}{判别式训练目标与 2.5D 输入}
  \begin{columns}[T]
    \column{0.52\textwidth}
      \begin{block}{模型思路}
        \begin{itemize}
          \item 2D 输入：仅使用当前 axial slice；
          \item 2.5D 输入：使用 $[x_{z-1},x_z,x_{z+1}]$ 三张相邻切片作为 3 通道；
          \item 输出仍为当前 slice 的四类分割 mask。
        \end{itemize}
      \end{block}
    \column{0.44\textwidth}
      \begin{exampleblock}{损失函数}
        使用 DiceCE 同时优化像素分类和区域重叠：
        \[
          \mathcal{L}
          =
          \mathcal{L}_{\mathrm{CE}}
          +
          \mathcal{L}_{\mathrm{Dice}}.
        \]
      \end{exampleblock}
      \sourcefoot{主要代码：\codepath{src/models/unet_baseline.py}, \codepath{src/models/segresnet_baseline.py}}
  \end{columns}
\end{frame}

\section{生成式模型}

\begin{frame}{生成式分割建模}
  \begin{columns}[T]
    \column{0.52\textwidth}
      \begin{block}{从确定性预测到条件分布}
        判别式模型通常输出一个确定 mask。生成式模型将 mask 视为条件随机变量：
        \[
          y\sim p_\theta(y\mid x).
        \]
        这样可以通过采样得到多个候选结果，并进一步分析模型不确定性。
      \end{block}
    \column{0.44\textwidth}
      \begin{exampleblock}{本项目生成式模型}
        \begin{itemize}
          \item conditional VAE；
          \item conditional diffusion；
        \end{itemize}
      \end{exampleblock}
  \end{columns}
\end{frame}

\begin{frame}{条件 VAE：潜变量生成分割掩码}
  \begin{columns}[T]
    \column{0.55\textwidth}
      \begin{block}{模型形式}
        给定 CT 图像 $x$ 和真实 mask $y$，编码器近似后验：
        \[
          q_\phi(z\mid x,y),
        \]
        解码器根据图像条件和潜变量生成 mask：
        \[
          \hat y\sim p_\theta(y\mid x,z).
        \]
      \end{block}
      \begin{exampleblock}{训练目标}
        \[
          \mathcal{L}_{\mathrm{cVAE}}
          =
          \mathcal{L}_{\mathrm{DiceCE}}(\hat y,y)
          +\beta D_{\mathrm{KL}}(q_\phi(z\mid x,y)\|p(z)).
        \]
      \end{exampleblock}
    \column{0.41\textwidth}
      \centering
      \includegraphics[width=0.92\textwidth]{assets/vae_cvae_diagram.png}
      \sourcefoot{cVAE 作为轻量生成式基线。}
      \begin{alertblock}{测试阶段}
        测试时真实 mask 不可见，只能从先验生成，因此 cVAE 对小目标更不稳定。
      \end{alertblock}
  \end{columns}
\end{frame}

\begin{frame}{条件 VAE：ELBO 推导}
  \small
  对本任务，条件是 CT 图像 $x$，要生成的是分割 mask $y$。目标是最大化条件似然 $\log p_\theta(y\mid x)$：
  \[
  \begin{aligned}
    \log p_\theta(y\mid x)
    &= \log \int p_\theta(y,z\mid x)\,dz \\
    &= \log \int
      \frac{q_\phi(z\mid x,y)}{q_\phi(z\mid x,y)}
      p_\theta(y\mid x,z)p(z)\,dz \\
    &\ge
      \mathbb{E}_{q_\phi(z\mid x,y)}
      \left[
        \log p_\theta(y\mid x,z)
      \right]
      -
      D_{\mathrm{KL}}\!\left(q_\phi(z\mid x,y)\|p(z)\right).
  \end{aligned}
  \]
  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{重建项}
        让生成的 mask $\hat y$ 接近真实标注 $y$，实现中用 DiceCE 近似负对数似然。
      \end{block}
    \column{0.48\textwidth}
      \begin{alertblock}{正则项}
        将后验 $q_\phi(z\mid x,y)$ 拉近先验 $p(z)=\mathcal{N}(0,I)$，使测试时可以从先验采样。
      \end{alertblock}
  \end{columns}
\end{frame}

\begin{frame}{条件扩散模型：从 noisy mask 到 clean mask}
  \small
  \begin{block}{前向加噪：只固定边缘分布}
    对真实 one-hot mask $m_0$ 加噪，并保持 $q(m_t\mid m_0)$ 的闭式形式：
    \[
      m_t=\sqrt{\bar{\alpha}_t}m_0+\sqrt{1-\bar{\alpha}_t}\epsilon,\quad
      \epsilon\sim\mathcal{N}(0,I).
    \]
  \end{block}
  \begin{columns}[T]
    \column{0.50\textwidth}
      \begin{exampleblock}{x0 条件去噪}
        \[
          \hat m_0=f_\theta(x,m_t,t).
        \]
        再由 $\hat m_0$ 反推出噪声方向：
        \[
          \hat\epsilon_t=
          \frac{m_t-\sqrt{\bar\alpha_t}\hat m_0}
          {\sqrt{1-\bar\alpha_t}}.
        \]
      \end{exampleblock}
    \column{0.46\textwidth}
      \begin{alertblock}{DDIM 跳步采样}
        DDIM 不要求沿完整马尔可夫链逐步反推，而是选择子序列 $t\to t'$：
        \[
          m_{t'}=
          \sqrt{\bar\alpha_{t'}}\hat m_0+
          \sqrt{1-\bar\alpha_{t'}}\hat\epsilon_t.
        \]
        这里等价于取 $\sigma_t=0$，采样过程确定，可用很少步数完成推理。
      \end{alertblock}
  \end{columns}
  \vspace{0.35em}
  \begin{block}{Setting}
    训练使用 $T=1000$ 的噪声时间步；测试时只取 1/5/10/25 个 DDIM 采样步进行比较。
  \end{block}
\end{frame}

\begin{frame}{x0 Diffusion：使生成目标对齐分割指标}
  \scriptsize
  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{传统 $\epsilon$ 预测的目标错位}
        若网络学习噪声，训练目标主要是：
        \[
          \mathcal{L}_{\epsilon}
          =
          \mathbb{E}_{t,\epsilon}
          \left[
            \|\epsilon_\theta(x,m_t,t)-\epsilon\|_2^2
          \right].
        \]
        它关注高斯噪声的数值误差；但最终评估发生在离散 mask 上：
        \[
          \hat y=\arg\max_c \hat m_{0,c},
          \qquad
          \mathrm{Dice}(\hat y,y),\ \mathrm{IoU}(\hat y,y).
        \]
        因此噪声 MSE 下降，不一定等价于 tumor/cyst 区域重叠变好。
      \end{block}
    \column{0.48\textwidth}
      \begin{alertblock}{x0 预测的目标对齐}
        本项目直接预测 clean mask logits：
        \[
          \hat m_0=f_\theta(x,m_t,t).
        \]
        这样可以直接在分割空间监督：
        \[
          \mathcal{L}_{x0}
          =
          \mathcal{L}_{\mathrm{CE}}(\hat m_0,y)
          +
          \mathcal{L}_{\mathrm{Dice}}(\hat m_0,y)
          +0.2\|\mathrm{softmax}(\hat m_0)-m_0\|_2^2.
        \]
        损失函数直接惩罚类别错误和区域重叠不足，与 Dice/IoU 的评价对象一致。
      \end{alertblock}
  \end{columns}
  \vspace{0.25em}
  \begin{exampleblock}{与 DDIM 采样兼容}
    即使网络预测的是 $\hat m_0$，仍可由
    $\hat\epsilon_t=(m_t-\sqrt{\bar\alpha_t}\hat m_0)/\sqrt{1-\bar\alpha_t}$
    得到噪声方向，再进行 DDIM 跳步更新。
  \end{exampleblock}
\end{frame}

      \begin{alertblock}{核心结论}
        \begin{itemize}
          \item 2D x0 diffusion 的 Mean Dice 为 0.434，明显优于 cVAE 的 0.201；
          \item 2.5D x0 diffusion 进一步提升到 0.534，说明相邻切片上下文对生成式模型有效；
          \item 判别式模型仍然更强，2.5D U-Net 达到 0.623；SegResNet 2.5D 已测试但未提升；
          \item 生成式模型的优势主要体现在生成过程和不确定性表达。
        \end{itemize}
      \end{alertblock}
  \end{columns}
\end{frame}

\begin{frame}{局限与后续方向}
  \begin{block}{主要局限}
    \begin{itemize}
      \item 2.5D diffusion 已明显优于 2D diffusion，但仍低于 2.5D U-Net；
      \item pixel-space mask diffusion 推理较慢，5 步约 67 ms/image；
      \item 当前只使用 100 例病例，未使用交叉验证、模型集成和强后处理。
    \end{itemize}
  \end{block}
  \begin{exampleblock}{可能提升方向}
    引入 2.5D/3D 条件编码、region-based label、前景区域裁剪、类别重加权 / focal loss，以及将 diffusion 用作判别式模型之后的 mask refinement。
  \end{exampleblock}
\end{frame}
