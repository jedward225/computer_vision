\section{任务背景}

\begin{frame}{任务定义：KiTS23 4 类语义分割}
  \begin{columns}[T]
    \column{0.54\textwidth}
      \begin{block}{输入与输出}
        \footnotesize
        给定腹部 CT axial slice $x\in\mathbb{R}^{1\times256\times256}$，预测逐像素类别：
        \[
          y_i\in\{0,1,2,3\}
          =\{\text{background},\text{kidney},\text{tumor},\text{cyst}\}.
        \]
        本项目报告 kidney、tumor、cyst 三个前景类别的 Dice / IoU，并计算前景 Mean Dice。
      \end{block}
      \begin{alertblock}{难点}
        \footnotesize
        kidney 区域较稳定，tumor/cyst 目标小、类别不均衡强；2D 切片缺少跨层连续性，小病灶最容易漏检。
      \end{alertblock}

    \column{0.42\textwidth}
      \begin{block}{数据划分}
        \centering\footnotesize
        \begin{tabular}{lrrr}
          \toprule
          Split & Cases & Slices & FG Slices \\
          \midrule
          Train & 70 & 6950 & 5585 \\
          Val   & 15 & 1624 & 1323 \\
          Test  & 15 & 1151 & 901 \\
          \bottomrule
        \end{tabular}
      \end{block}
      \begin{exampleblock}{CT 预处理}
        \footnotesize
        HU window: $c=40,w=400$；线性归一化到 $[0,1]$；resize 到 $256\times256$；按 patient-level split 防止数据泄漏。
      \end{exampleblock}
  \end{columns}
\end{frame}

\begin{frame}{预处理与训练约束}
  \begin{columns}[T]
    \column{0.50\textwidth}
      \begin{block}{数据预处理}
        \footnotesize
        \begin{itemize}
          \item 从 NIfTI affine 推断 axial 轴，避免把 sagittal/coronal 切片误当 axial。
          \item CT 强度使用腹部窗口 $c=40,w=400$，再归一化到 $[0,1]$。
          \item 将切片 resize 到 $256\times256$，标签保留 4 类互斥类别。
          \item 使用 patient-level split，防止同一病例切片同时进入训练和测试集。
        \end{itemize}
      \end{block}
    \column{0.46\textwidth}
      \begin{block}{训练设置}
        \centering\tiny
        \setlength{\tabcolsep}{2.2pt}
        \begin{tabular}{@{}lccc@{}}
          \toprule
          Model & Ep. & BS & LR \\
          \midrule
          U-Net/SegResNet & 100 & 32 & $2e{-4}$ \\
          cVAE & 100 & 32 & $2e{-4}$ \\
          x0 Diffusion & 80 & 16 & $1e{-4}$ \\
          \bottomrule
        \end{tabular}
      \end{block}
      \begin{alertblock}{统一约束}
        \footnotesize
        模型选择依据 validation Mean Dice；diffusion 验证使用 5-step DDIM；最终测试统一在 axial test split 上评估。
      \end{alertblock}
  \end{columns}
\end{frame}

\begin{frame}{项目目标：判别式基线与条件生成式分割}
  \begin{columns}[T]
    \column{0.47\textwidth}
      \begin{exampleblock}{判别式 baseline}
        \footnotesize
        直接学习映射
        \begin{align*}
          \hat{y} &= f_\theta(x),\\
          p_\theta(y_i=c\mid x)
          &= \mathrm{softmax}(f_\theta(x)_i)_c .
        \end{align*}
        优点是目标直接、推理快、医学分割中很稳定。我们训练 U-Net、SegResNet 和 2.5D U-Net 作为对照。
      \end{exampleblock}
    \column{0.49\textwidth}
      \begin{alertblock}{生成式方案}
        \footnotesize
        将 mask 当作生成目标，学习条件分布
        \[
          p_\theta(y\mid x).
        \]
        生成式部分实现 conditional VAE 与 conditional diffusion：它们不只给出单个 mask，还可以展示生成轨迹和不确定性。
      \end{alertblock}
  \end{columns}
  \begin{block}{本项目核心问题}
    \footnotesize
    生成式模型能否完成医学分割？相比判别式模型，它的精度、速度、采样步数和不确定性表达分别如何？
  \end{block}
\end{frame}

\begin{frame}{实验方案概览}
  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{判别式模型}
        \footnotesize
        \begin{itemize}
          \item \model{U-Net}: encoder-decoder + skip connection
          \item \model{SegResNet}: residual blocks for segmentation
          \item \model{2.5D U-Net}: 相邻切片作为 3 通道输入
        \end{itemize}
      \end{block}
      \sourcefoot{主要代码：\codepath{src/models/unet_baseline.py}, \codepath{src/models/segresnet_baseline.py}}
    \column{0.48\textwidth}
      \begin{block}{生成式模型}
        \footnotesize
        \begin{itemize}
          \item \model{Conditional VAE}: $z$ 表示掩码生成的不确定性
          \item \model{x0 Diffusion}: 从 noisy mask 逐步生成 clean mask
          \item DDIM sampling: 比较 1/5/10/25 步
        \end{itemize}
      \end{block}
      \sourcefoot{主要代码：\codepath{src/models/cvae_seg.py}, \codepath{src/models/diffusion_unet.py}, \codepath{src/diffusion/scheduler.py}}
  \end{columns}
\end{frame}

\section{判别式模型}

\begin{frame}{判别式模型：方法方案与结果}
  \begin{columns}[T]
    \column{0.52\textwidth}
      \begin{block}{U-Net / SegResNet}
        \footnotesize
        判别式方法直接优化分割目标，训练损失使用 DiceCE：
        \[
          \mathcal{L}_{\mathrm{DiceCE}}
          =\mathcal{L}_{\mathrm{CE}}
          +\left(1-\frac{2\sum_i p_i g_i+\epsilon}{\sum_i p_i+\sum_i g_i+\epsilon}\right).
        \]
        CE 保证像素级分类，Dice 项缓解前景类别稀疏问题。
      \end{block}
    \column{0.44\textwidth}
      \begin{exampleblock}{测试集结果}
        \centering\tiny
        \setlength{\tabcolsep}{2.1pt}
        \begin{tabular}{@{}lcccc@{}}
          \toprule
          Model & Mean & Kid. & Tum. & Cyst \\
          \midrule
          U-Net 2D & 0.570 & 0.912 & 0.623 & 0.177 \\
          SegResNet & 0.572 & 0.914 & 0.621 & 0.181 \\
          U-Net 2.5D & \metricup{0.623} & 0.921 & 0.707 & 0.242 \\
          \bottomrule
        \end{tabular}
      \end{exampleblock}
      \begin{alertblock}{对照意义}
        \footnotesize
        判别式模型提供精度和速度参照。生成式模型需要同时说明分割质量、推理成本，以及额外的不确定性表达能力。
      \end{alertblock}
  \end{columns}
\end{frame}

\section{生成式模型}

\begin{frame}[shrink=8]{Conditional VAE：把 mask 当作条件生成目标}
  \begin{columns}[T]
    \column{0.55\textwidth}
      \begin{block}{条件生成建模}
        \scriptsize
        分割中 $x$ 是 CT 条件，$y$ 是真实 mask。CVAE 通过 latent variable $z$ 建模：
        \[
          q_\phi(z\mid x,y),\qquad p_\theta(y\mid x,z).
        \]
        条件 ELBO 为
        \[
          \log p_\theta(y\mid x)\ge
          \mathbb{E}_{q_\phi}[\log p_\theta(y\mid x,z)]
          -D_{\mathrm{KL}}(q_\phi(z\mid x,y)\|p(z)).
        \]
      \end{block}
      \begin{exampleblock}{本项目损失}
        \scriptsize
        \[
          \mathcal{L}_{\mathrm{cVAE}}
          =
          \mathcal{L}_{\mathrm{DiceCE}}(\hat y,y)
          +\beta D_{\mathrm{KL}}(q_\phi(z\mid x,y)\|p(z)).
        \]
      \end{exampleblock}
    \column{0.41\textwidth}
      \begin{alertblock}{cVAE 的局限}
        \scriptsize
        训练时 encoder 看到真实 mask；测试时真实 $y$ 不可见，只能从 prior $z\sim p(z)$ 或 $z=0$ 解码。这个 posterior-prior gap 会导致小目标生成不稳定。
      \end{alertblock}
      \centering
      \includegraphics[width=0.92\textwidth]{assets/vae_cvae_diagram.jpg}
  \end{columns}
\end{frame}

\begin{frame}[shrink=8]{DDPM / DDIM：从 noisy mask 逐步去噪}
  \begin{block}{扩散前向过程}
    \scriptsize
    对真实 one-hot mask $m_0$ 加高斯噪声：
    \[
      q(m_t\mid m_0)=
      \mathcal{N}(\sqrt{\bar{\alpha}_t}m_0,(1-\bar{\alpha}_t)I),
      \quad
      m_t=\sqrt{\bar{\alpha}_t}m_0+\sqrt{1-\bar{\alpha}_t}\epsilon.
    \]
  \end{block}
  \begin{columns}[T]
    \column{0.50\textwidth}
      \begin{exampleblock}{标准 DDPM 目标}
        \scriptsize
        \[
          \mathcal{L}_{\epsilon}
          =
          \|\epsilon-\epsilon_\theta(x,m_t,t)\|_2^2.
        \]
        这个目标在自然图像生成中常用，但在医学分割中容易与 Dice 指标不对齐。
      \end{exampleblock}
    \column{0.46\textwidth}
      \begin{alertblock}{DDIM 采样}
        \scriptsize
        测试时不走完整 1000 步，而是在较短时间序列上采样。我们比较 $S=1,5,10,25$ 步，观察精度与速度 trade-off。
      \end{alertblock}
  \end{columns}
  \centering
  \includegraphics[width=0.58\textwidth]{assets/ddpm_ddim_diagram.png}
\end{frame}

\begin{frame}{x0 Diffusion：生成目标设计}
  \begin{columns}[T]
    \column{0.54\textwidth}
      \begin{block}{从 epsilon 到 x0}
        \footnotesize
        早期实验使用 $\epsilon$ prediction，train loss 可以下降，但 validation Dice 长期很低。原因是 MSE 主要被背景和连续噪声主导，不直接优化 mask 结构。
      \end{block}
      \begin{exampleblock}{最终主模型：直接预测 clean mask}
        \footnotesize
        网络输入 CT、noisy mask 和 timestep，直接输出 $\hat m_0$ logits：
        \[
          \hat m_0=f_\theta(x,m_t,t).
        \]
        \[
          \mathcal{L}_{x0}
          =
          \mathcal{L}_{\mathrm{DiceCE}}(\hat m_0,m_0)
          +0.2\|\mathrm{softmax}(\hat m_0)-m_0\|_2^2.
        \]
      \end{exampleblock}
    \column{0.42\textwidth}
      \begin{alertblock}{关键结论}
        \footnotesize
        x0 目标把训练信号直接放在 segmentation mask 上，因此比标准噪声预测更适合 dense prediction。
      \end{alertblock}
      \begin{block}{配置}
        \centering\scriptsize
        \begin{tabular}{lr}
          \toprule
          Item & Setting \\
          \midrule
          Train epochs & 80 \\
          Train steps & 1000 \\
          Validation sample steps & 5 \\
          Batch size & 16 \\
          LR & $10^{-4}$ \\
          \bottomrule
        \end{tabular}
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{条件 U-Net 去噪网络}
  \begin{columns}[T]
    \column{0.52\textwidth}
      \begin{block}{网络输入}
        \footnotesize
        \codepath{src/models/diffusion_unet.py}\\[2pt]
        stem 输入通道为 image channels + mask channels：
        \[
          [x,m_t]\in\mathbb{R}^{(1+4)\times256\times256}.
        \]
        每个 residual block 接收 sinusoidal timestep embedding：
        \[
          h_{l+1}=\mathrm{ResBlock}(h_l,\mathrm{MLP}(\gamma(t))).
        \]
      \end{block}
      \begin{exampleblock}{推理流程}
        \footnotesize
        从随机 mask logits $m_T$ 出发，用 DDIM 逐步得到 $\hat m_0$，最后 argmax 得到 4-class segmentation。
      \end{exampleblock}
    \column{0.44\textwidth}
      \centering
      \includegraphics[width=\textwidth]{assets/axial_diffusion_denoising.png}
      \sourcefoot{x0 diffusion denoising trajectory, 5 steps.}
  \end{columns}
\end{frame}

      \begin{exampleblock}{有效的地方}
        \footnotesize

        \footnotesize

\begin{frame}{采样步数消融：精度与速度权衡}
  \begin{columns}[T]
    \column{0.46\textwidth}
      \begin{block}{x0 diffusion test set}
        \centering\scriptsize
        \begin{tabular}{rrrr}
          \toprule
          Steps & Mean Dice & Mean IoU & ms/img \\
          \midrule
          1  & 0.415 & 0.320 & 13.60 \\
          5  & \metricup{0.434} & \metricup{0.334} & 67.43 \\
          10 & 0.413 & 0.316 & 134.85 \\
          25 & 0.418 & 0.317 & 337.09 \\
          \bottomrule
        \end{tabular}
      \end{block}
      \begin{alertblock}{结论}
        \footnotesize
        5 步是本实验中的最佳折中。10/25 步没有提升 Dice，却几乎线性增加推理时间。
      \end{alertblock}
    \column{0.50\textwidth}
      \begin{exampleblock}{解释}
        \footnotesize
        x0 diffusion 在本实验设置下更接近“带 noisy mask 条件的少步分割器”。过多迭代可能在 tumor/cyst 小区域累积误差。
      \end{exampleblock}
      \begin{block}{机制解释}
        \footnotesize
        生成式分割的瓶颈不只是采样步数，而是目标设计、类别不均衡和条件编码能力。
      \end{block}
  \end{columns}
\end{frame}

\section{总结}

\begin{frame}{本项目结论}
  \begin{columns}[T]
    \column{0.49\textwidth}
      \begin{exampleblock}{我们做到了什么}
        \footnotesize
        \begin{itemize}
          \item 将 KiTS23 4-class segmentation 建模为 conditional mask generation
          \item 实现 cVAE 与 x0 diffusion，并与 U-Net / SegResNet 对比
          \item 做了采样步数消融、denoising trajectory 和 uncertainty visualization
        \end{itemize}
      \end{exampleblock}
    \column{0.49\textwidth}
      \begin{alertblock}{结果说明什么}
        \footnotesize
        \begin{itemize}
          \item x0 diffusion test Mean Dice 0.434，明显优于 cVAE 0.201
          \item 判别式模型仍更强更快，2.5D U-Net 达到 0.623
          \item 生成式模型的价值在轨迹、不确定性和分布建模，而不是单点 Dice 最优
        \end{itemize}
      \end{alertblock}
  \end{columns}
\end{frame}

\begin{frame}{局限与下一步}
  \begin{block}{主要局限}
    \footnotesize
    \begin{itemize}
      \item 2D diffusion 缺少完整 3D 上下文；tumor/cyst 小目标仍不稳定
      \item pixel-space mask diffusion 推理慢，5 steps 已约 67 ms/image
      \item 标准 epsilon prediction 与 segmentation metric 不对齐，必须改成 x0 目标
    \end{itemize}
  \end{block}
  \begin{exampleblock}{可继续提升的方向}
    \footnotesize
    2.5D/3D conditional encoder；region-based label；前景区域裁剪；类别重加权 / focal loss；将 diffusion 用作判别式模型之后的 mask refinement，而不是从随机 mask 完整生成。
  \end{exampleblock}
\end{frame}
